# Всплывающие сообщения — план

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Отказы файлам и предупреждения о составе уезжают из потока
страницы во всплывающие окошки в правом нижнем углу.

**Architecture:** Новый модуль `toasts.js` — владелец узла, устроенный как
`tips.js`: свой контейнер, свои таймеры, наружу одна функция `show`.
Окошки строятся узлами через `createElement`, а не строкой в `innerHTML`,
поэтому текст причины не может стать разметкой. Звонят в него `files.js`
(отказ файлу) и `ui.js` (новые строки `store.warnings()`).

**Tech Stack:** Ванильный ES2015+ в UMD-обёртке вокруг `KP`, CSS без
препроцессора, сборка одним файлом через `dashboard/build.py`, тесты —
`node --test` поверх `tests/js/domstub.js`.

## Global Constraints

- Собранная страница остаётся одним самодостаточным файлом: никаких
  внешних запросов, никаких библиотек.
- Формат снапшота (`schema: 1`), ключи в адресной строке и золотая фикстура
  `tests/js/fixtures/page-data.golden.json` не трогаются.
- Тексты отказов из `store.js` и `files.js` остаются дословно теми же.
- Отклик «Copy NVR» (подпись кнопки на 1.4 с) не трогается.
- Класс `.mark.warn` в `table.css` не трогается: он метит изменившуюся
  ветку и к страничным предупреждениям отношения не имеет.
- Комментарии и подписи — по-русски, в тон соседним файлам: объясняют
  «почему», а не пересказывают код.
- Номер версии не поднимается: 2.0.0 ещё не выпущена, дописывается её же
  запись в `CHANGELOG.md`.
- Ветка `feature/toasts` от `develop`; в `develop` не вливать без апрува
  человека.
- После каждой задачи обе сюиты зелёные:
  `python3 -m unittest discover -s tests` и `node --test tests/js/*.test.js`.

### Task 1: Модуль `toasts.js`

**Files:**
- Create: `dashboard/assets/js/toasts.js`
- Create: `tests/js/toasts.test.js`
- Modify: `dashboard/build.py:20-23` (кортеж `SCRIPTS`)

**Interfaces:**
- Consumes: ничего. Модуль ни от кого не зависит — как `tips.js`.
- Produces: `KP.toasts.create({ node, clock }) -> { show }`, где
  - `node` — узел-контейнер, в него окошки кладутся `appendChild`;
  - `clock` — необязательная пара `{ set, clear }`; по умолчанию
    `setTimeout`/`clearTimeout`;
  - `show({ kind, title, lines })` — `kind` это `'error'` или `'warn'`
    (что угодно другое считается `'warn'`), `title` необязателен,
    `lines` — массив строк; пустой массив ничего не показывает.
    Возвращает `undefined`.

'use strict';
var test = require('node:test');
var assert = require('node:assert');
var domstub = require('./domstub');
var toastsmod = require('../../dashboard/assets/js/toasts');

/* Часы теста: модуль просит время у них, а не у node, поэтому «прошло
   десять секунд» здесь стоит одной строкой и правда мгновенно. */
function clock() {
  var pending = [], next = 1;
  return {
    set: function (fn, ms) {
      pending.push({ id: next, fn: fn, at: ms });
      return next++;
    },
    clear: function (id) {
      pending = pending.filter(function (t) { return t.id !== id; });
    },
    /* Стреляем всем, чей срок не больше прошедшего: два таймера окошка
       заведены на LIFE и на LIFE + FADE, и тесту нужно уметь дойти до
       каждого по отдельности. */
    tick: function (ms) {
      var due = pending.filter(function (t) { return t.at <= ms; });
      pending = pending.filter(function (t) { return t.at > ms; })
        .map(function (t) { return { id: t.id, fn: t.fn, at: t.at - ms }; });
      due.forEach(function (t) { t.fn(); });
    },
    count: function () { return pending.length; }
  };
}

function setup() {
  var dom = domstub.install();
  var box = dom.document.createElement('div');

  return { dom: dom, box: box, clk: clk,

Создать `dashboard/assets/js/toasts.js`:

```js
/* Всплывающие сообщения: окошки в правом нижнем углу, которые гаснут
   сами. Владеет одним контейнером на всю страницу и таймерами своих
   окошек; наружу отдаёт только show.

   Сообщение о событии не имеет права двигать страницу: человек в этот
   момент читает таблицу, а строка, которая появляется и через десять
   секунд исчезает, увозит её сначала вниз, а потом обратно. Окошко лежит
   поверх страницы и не двигает ничего.

   Окошки собираются узлами, а не строкой в innerHTML: в причине отказа
   стоит имя файла, пришедшее снаружи, и textContent делает его текстом
   по устройству, а не по дисциплине вызывающего. */
(function (root, factory) {
  if (typeof module === 'object' && module.exports) {
    module.exports = factory();
  } else {
    root.KP = root.KP || {};
    root.KP.toasts = factory();
  }
}(typeof globalThis !== 'undefined' ? globalThis : this, () => {
  'use strict';

  /* Десять секунд — столько, сколько сообщение о загрузке жило строкой:
     хватает прочитать несколько строк, а не хватит — под курсором отсчёт
     останавливается. */
  const LIFE = 10000;
  const FADE = 400;
  /* Угол экрана — не лента: стопка в полэкрана закрывала бы страницу
     вместо того, чтобы о ней рассказывать. */
  const MAX_TOASTS = 4;
  const MAX_LINES = 6;

  function create(deps) {
    const box = deps.node;
    const clock = deps.clock
      || { set: (fn, ms) => setTimeout(fn, ms), clear: (id) => clearTimeout(id) };
    /* Живые окошки в порядке появления: старшее — первое. */
    let live = [];

    function timer(fn, ms) {
      const id = clock.set(fn, ms);
      /* В node таймер держит процесс живым, и сюита ждала бы срока
         каждого окошка по-настоящему. В браузере setTimeout возвращает
         число, у которого unref нет, и строка ничего не делает. */
      if (id && id.unref) id.unref();
      return id;
    }

    function stop(toast) {
      clock.clear(toast.dim);
      clock.clear(toast.gone);
      toast.dim = null;
      toast.gone = null;
    }

    function drop(toast) {
      stop(toast);
      const at = live.indexOf(toast);
      if (at !== -1) live.splice(at, 1);
      if (toast.el.parentNode) toast.el.parentNode.removeChild(toast.el);
    }

    function fade(toast) {
      toast.el.className = `toast ${toast.kind} fading`;
      toast.gone = timer(() => drop(toast), FADE);
    }

    /* Курсор пришёл: отсчёт стоит, и класс гашения снимается сразу — если
       окошко уже догорало, оно возвращается к жизни, а не остаётся
       прозрачным до ухода курсора. */
    function hold(toast) {
      stop(toast);
      toast.el.className = `toast ${toast.kind}`;
    }

    /* Отсчёт всегда начинается с полного срока: окошко, к которому
       вернулись, не должно гаснуть быстрее того, которое видят впервые. */
    function arm(toast) {
      hold(toast);
      toast.dim = timer(() => fade(toast), LIFE);
    }

    function el(tag, cls, text) {
      const node = document.createElement(tag);
      if (cls) node.className = cls;
      if (text !== undefined) node.textContent = text;
      return node;
    }

    function listOf(lines) {
      const list = el('ul', 'toast-l');
      /* Режем, только если под «…и ещё N» уходит больше одной строки:
         иначе он занимает ровно место строки и вместо причины сообщает
         о ней. */
      const cut = lines.length > MAX_LINES + 1 ? MAX_LINES : lines.length;
      for (let i = 0; i < cut; i++) list.appendChild(el('li', '', lines[i]));
      if (cut < lines.length) {
        list.appendChild(el('li', 'more', `…и ещё ${lines.length - cut}`));
      }
      return list;
    }

    function show(spec) {
      const lines = (spec && spec.lines) || [];
      if (!lines.length) return;
      const kind = spec.kind === 'error' ? 'error' : 'warn';
      const wrap = el('div', `toast ${kind}`);
      const kill = el('button', 'toast-x', '✕');
      kill.setAttribute('type', 'button');
      kill.setAttribute('aria-label', 'Закрыть');
      wrap.appendChild(kill);
      if (spec.title) wrap.appendChild(el('div', 'toast-t', spec.title));
      wrap.appendChild(listOf(lines));

      const toast = { el: wrap, kind: kind, dim: null, gone: null };
      kill.addEventListener('click', () => drop(toast));
      wrap.addEventListener('mouseenter', () => hold(toast));
      wrap.addEventListener('mouseleave', () => arm(toast));
      /* До крестика доходят и табом: под клавиатурой отсчёт держим так
         же, как под мышью. */
      wrap.addEventListener('focusin', () => hold(toast));
      wrap.addEventListener('focusout', () => arm(toast));

      box.appendChild(wrap);
      live.push(toast);
      /* Лишнее окошко выталкивается сразу, не дожидаясь своего срока. */
      while (live.length > MAX_TOASTS) drop(live[0]);
      arm(toast);
    }

    return { show };
  }

  return { create };
}));
```
